cmake_minimum_required(VERSION 3.24)
project(stdio.sys VERSION 0.1.0 LANGUAGES C)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "stdio.sys requires MSVC targeting Windows x64")
endif()
if(CMAKE_GENERATOR_PLATFORM AND NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "Select the x64 platform")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /MACHINE:X64)

add_library(engine STATIC src/tree.c src/shell.c src/index.c)
target_link_libraries(engine PUBLIC ntdll shell32 ole32 uuid advapi32)
add_executable(stdio WIN32 src/main.c)
target_link_libraries(stdio PRIVATE engine)
add_library(stdio_extension SHARED src/explorer.c src/explorer.def)
set_target_properties(stdio_extension PROPERTIES OUTPUT_NAME stdio ARCHIVE_OUTPUT_NAME stdio_extension)
target_link_libraries(stdio_extension PRIVATE shell32 shlwapi ole32 uuid advapi32)
add_executable(stdioctl src/ctl.c src/bench.c)
target_link_libraries(stdioctl PRIVATE engine fltlib psapi bcrypt)
install(TARGETS stdio stdio_extension stdioctl RUNTIME DESTINATION . LIBRARY DESTINATION .)

include(CTest)
option(STDIO_SHELL_LAB "Build the isolated Shell properties experiment" OFF)
if(STDIO_SHELL_LAB)
    add_library(stdio_shell_lab SHARED tests/shell_stream_lab.c src/folder_streams.c)
    target_include_directories(stdio_shell_lab PRIVATE src)
    target_link_libraries(stdio_shell_lab PRIVATE bcrypt shell32 onecore user32)
endif()
if(BUILD_TESTING)
    add_executable(stdio_smoke tests/smoke.c)
    target_link_libraries(stdio_smoke PRIVATE engine)
    add_test(NAME stdio.smoke COMMAND stdio_smoke "$<TARGET_FILE:stdio_extension>")
    add_executable(stdio_folder_streams tests/folder_streams.c src/folder_streams.c)
    target_include_directories(stdio_folder_streams PRIVATE src)
    add_test(NAME stdio.folder_streams COMMAND stdio_folder_streams)
    add_test(NAME stdio.index COMMAND powershell.exe -NoLogo -NoProfile -NonInteractive
             -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/index.ps1"
             "$<TARGET_FILE:stdioctl>" "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(stdio.index PROPERTIES TIMEOUT 30)
endif()
